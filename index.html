using UnityEngine; using UnityEngine.SceneManagement; using System.Collections.Generic; public class AdityaBattleArena : MonoBehaviour { class Enemy { public GameObject obj; public int hp = 100; public float timer; } CharacterController controller; GameObject player; Camera cam; List enemies = new List(); List objects = new List(); float health = 100f; float yaw, pitch = 12f, verticalSpeed; float fireCooldown, reloadTimer; int ammo = 30, reserveAmmo = 120, kills; bool gameOver, won; void Start() { CreateWorld(); CreatePlayer(); CreateCamera(); CreateEnemies(); Cursor.lockState = CursorLockMode.Locked; Cursor.visible = false; } void Update() { if (gameOver) { if (Input.GetKeyDown(KeyCode.Return)) SceneManager.LoadScene(SceneManager.GetActiveScene().buildIndex); return; } if (Input.GetKeyDown(KeyCode.Escape)) { Cursor.lockState = CursorLockMode.None; Cursor.visible = true; } if (Input.GetMouseButtonDown(0) && Cursor.lockState != CursorLockMode.Locked) { Cursor.lockState = CursorLockMode.Locked; Cursor.visible = false; } Look(); MovePlayer(); UpdateCamera(); if (Input.GetMouseButton(0) || Input.GetKeyDown(KeyCode.Space)) Shoot(); if (Input.GetKeyDown(KeyCode.R)) Reload(); if (fireCooldown > 0) fireCooldown -= Time.deltaTime; if (reloadTimer > 0) { reloadTimer -= Time.deltaTime; if (reloadTimer <= 0) { int amount = Mathf.Min(30 - ammo, reserveAmmo); ammo += amount; reserveAmmo -= amount; } } UpdateEnemies(); if (health <= 0) { health = 0; gameOver = true; won = false; } else if (enemies.Count == 0) { gameOver = true; won = true; } } void CreateWorld() { RenderSettings.ambientLight = new Color(.65f, .68f, .72f); RenderSettings.fog = true; RenderSettings.fogColor = new Color(.65f, .78f, .72f); RenderSettings.fogStartDistance = 65; RenderSettings.fogEndDistance = 150; GameObject ground = GameObject.CreatePrimitive(PrimitiveType.Plane); ground.name = "Battle Island"; ground.transform.localScale = new Vector3(12, 1, 12); ground.GetComponent().material.color = new Color(.28f, .48f, .22f); GameObject sun = new GameObject("Sun"); Light light = sun.AddComponent(); light.type = LightType.Directional; light.intensity = 1.2f; sun.transform.rotation = Quaternion.Euler(48, -35, 0); for (int i = 0; i < 12; i++) { Vector3 p = RandomPosition(14); GameObject house = GameObject.CreatePrimitive(PrimitiveType.Cube); house.name = "House"; house.transform.position = p + Vector3.up * 2; house.transform.localScale = new Vector3(5, 4, 5); house.GetComponent().material.color = new Color(.58f, .39f, .23f); objects.Add(house); GameObject roof = GameObject.CreatePrimitive(PrimitiveType.Cube); roof.name = "Roof"; roof.transform.position = p + Vector3.up * 4.5f; roof.transform.localScale = new Vector3(5.8f, 1, 5.8f); roof.GetComponent().material.color = new Color(.35f, .16f, .12f); objects.Add(roof); } for (int i = 0; i < 45; i++) { Vector3 p = RandomPosition(12); GameObject trunk = GameObject.CreatePrimitive(PrimitiveType.Cylinder); trunk.name = "Tree Trunk"; trunk.transform.position = p + Vector3.up * 1.5f; trunk.transform.localScale = new Vector3(.5f, 1.5f, .5f); trunk.GetComponent().material.color = new Color(.3f, .18f, .08f); GameObject leaves = GameObject.CreatePrimitive(PrimitiveType.Sphere); leaves.name = "Tree Leaves"; leaves.transform.position = p + Vector3.up * 4; leaves.transform.localScale = new Vector3(3, 4, 3); leaves.GetComponent().material.color = new Color(.12f, .38f, .12f); objects.Add(trunk); objects.Add(leaves); } } Vector3 RandomPosition(float safeDistance) { Vector3 p; do { p = new Vector3(Random.Range(-52f, 52f), 0, Random.Range(-52f, 52f)); } while (p.magnitude < safeDistance); return p; } void CreatePlayer() { player = new GameObject("Aditya Player"); player.transform.position = new Vector3(0, 1, 0); controller = player.AddComponent(); controller.height = 2; controller.radius = .4f; controller.center = Vector3.zero; GameObject body = GameObject.CreatePrimitive(PrimitiveType.Capsule); body.name = "Player Body"; body.transform.SetParent(player.transform, false); body.GetComponent().enabled = false; body.GetComponent().material.color = new Color(.1f, .35f, .95f); GameObject head = GameObject.CreatePrimitive(PrimitiveType.Sphere); head.name = "Player Head"; head.transform.SetParent(player.transform, false); head.transform.localPosition = new Vector3(0, .72f, 0); head.transform.localScale = Vector3.one * .48f; head.GetComponent().enabled = false; head.GetComponent().material.color = new Color(.92f, .68f, .48f); GameObject gun = GameObject.CreatePrimitive(PrimitiveType.Cube); gun.name = "Rifle"; gun.transform.SetParent(player.transform, false); gun.transform.localPosition = new Vector3(.35f, .25f, .55f); gun.transform.localScale = new Vector3(.16f, .16f, .9f); gun.GetComponent().enabled = false; gun.GetComponent().material.color = Color.black; } void CreateCamera() { cam = Camera.main; if (cam == null) { GameObject go = new GameObject("Main Camera"); cam = go.AddComponent(); cam.tag = "MainCamera"; } cam.fieldOfView = 65; cam.nearClipPlane = .1f; cam.farClipPlane = 250; if (cam.GetComponent() == null) cam.gameObject.AddComponent(); UpdateCamera(); } void CreateEnemies() { for (int i = 0; i < 8; i++) { GameObject obj = GameObject.CreatePrimitive(PrimitiveType.Capsule); obj.name = "Enemy Soldier " + (i + 1); obj.transform.position = RandomPosition(17) + Vector3.up; obj.GetComponent().material.color = new Color(.85f, .12f, .12f); enemies.Add(new Enemy { obj = obj, hp = 100, timer = Random.Range(1f, 3f) }); } } void Look() { if (Cursor.lockState != CursorLockMode.Locked) return; yaw += Input.GetAxis("Mouse X") * 3f; pitch -= Input.GetAxis("Mouse Y") * 2f; pitch = Mathf.Clamp(pitch, -15f, 55f); } void MovePlayer() { float x = Input.GetAxisRaw("Horizontal"); float z = Input.GetAxisRaw("Vertical"); Vector3 forward = Quaternion.Euler(0, yaw, 0) * Vector3.forward; Vector3 right = Quaternion.Euler(0, yaw, 0) * Vector3.right; Vector3 move = forward * z + right * x; if (move.magnitude > 1) move.Normalize(); if (controller.isGrounded && verticalSpeed < 0) verticalSpeed = -2; if (Input.GetKeyDown(KeyCode.Space) && controller.isGrounded) verticalSpeed = 7; verticalSpeed += -22f * Time.deltaTime; move.y = verticalSpeed; controller.Move(move * 6f * Time.deltaTime); if (x != 0 || z != 0) player.transform.rotation = Quaternion.Euler(0, yaw, 0); } void UpdateCamera() { if (cam == null || player == null) return; Quaternion rot = Quaternion.Euler(pitch, yaw, 0); Vector3 target = player.transform.position + Vector3.up * 1.2f; cam.transform.position = target + rot * new Vector3(0, 1.4f, -5.5f); cam.transform.LookAt(target + rot * Vector3.forward * 8); } void Shoot() { if (reloadTimer > 0 || ammo <= 0 || fireCooldown > 0) return; fireCooldown = .18f; ammo--; Ray ray = cam.ViewportPointToRay(new Vector3(.5f, .5f, 0)); RaycastHit hit; if (Physics.Raycast(ray, out hit, 150f)) { for (int i = enemies.Count - 1; i >= 0; i--) { Enemy e = enemies[i]; if (hit.transform.gameObject == e.obj) { e.hp -= 50; if (e.hp <= 0) { Destroy(e.obj); enemies.RemoveAt(i); kills++; } break; } } } } void Reload() { if (reloadTimer <= 0 && ammo < 30 && reserveAmmo > 0) reloadTimer = 1.5f; } void UpdateEnemies() { foreach (Enemy e in enemies) { if (e.obj == null) continue; Vector3 direction = player.transform.position - e.obj.transform.position; float distance = direction.magnitude; if (distance > 2.5f) { direction.y = 0; if (direction.sqrMagnitude > .01f) { e.obj.transform.position += direction.normalized * 2f * Time.deltaTime; e.obj.transform.rotation = Quaternion.LookRotation(direction.normalized); } } e.timer -= Time.deltaTime; if (distance < 28f && e.timer <= 0) { e.timer = Random.Range(1f, 2.2f); health -= Random.Range(3f, 7f); } } } void OnGUI() { GUIStyle style = new GUIStyle(GUI.skin.label); style.fontSize = 20; style.fontStyle = FontStyle.Bold; style.normal.textColor = Color.white; GUI.Label(new Rect(15, 12, 350, 30), "ADITYA BATTLE ARENA", style); GUI.Label(new Rect(15, 45, 250, 30), "Health: " + Mathf.CeilToInt(health), style); GUI.Label(new Rect(15, 77, 250, 30), "Kills: " + kills + " / 8", style); GUI.Label(new Rect(15, 109, 350, 30), "Ammo: " + ammo + " | Reserve: " + reserveAmmo, style); GUI.Label(new Rect(Screen.width / 2 - 10, Screen.height / 2 - 15, 30, 30), "+", style); GUI.Label(new Rect(15, Screen.height - 35, 850, 30), "WASD: Move | Mouse: Aim | Left Click: Shoot | R: Reload | Space: Jump", style); if (reloadTimer > 0) GUI.Label(new Rect(Screen.width / 2 - 70, 65, 200, 30), "RELOADING...", style); if (gameOver) { GUI.Box(new Rect(Screen.width / 2 - 170, Screen.height / 2 - 65, 340, 130), ""); GUI.Label(new Rect(Screen.width / 2 - 120, Screen.height / 2 - 40, 300, 35), won ? "VICTORY! YOU WIN!" : "GAME OVER", style); GUI.Label(new Rect(Screen.width / 2 - 120, Screen.height / 2, 300, 30), "Press ENTER to restart", style); } } }
